cmake_minimum_required(VERSION 3.13)
project(LogHarbor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB_RECURSE LOGHARBOR_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(logharbor ${LOGHARBOR_SOURCES})
target_include_directories(logharbor PUBLIC include)
foreach(tool logscan logindex logquery logredact)
  add_executable(${tool} tools/${tool}.cpp)
  target_link_libraries(${tool} PRIVATE logharbor)
endforeach()
enable_testing()
file(GLOB_RECURSE TEST_SOURCES CONFIGURE_DEPENDS tests/*.cpp)
foreach(test_src ${TEST_SOURCES})
  get_filename_component(test_name ${test_src} NAME_WE)
  add_executable(${test_name} ${test_src})
  target_link_libraries(${test_name} PRIVATE logharbor)
  add_test(NAME ${test_name} COMMAND ${test_name})
endforeach()
option(LOGHARBOR_BUILD_FUZZERS "Build local fuzz targets" OFF)
if(LOGHARBOR_BUILD_FUZZERS)
  foreach(fz ingest_fuzzer index_fuzzer query_fuzzer stream_fuzzer)
    add_executable(${fz} fuzz/${fz}.cc)
    target_link_libraries(${fz} PRIVATE logharbor)
  endforeach()
endif()
